   📋 Код метода:
   def start_adjustment(self):
           """
           Начало настройки.
           
           Для гармонического режима: тоника звучит как фон, интервал настраивается.
           Для мелодического режима: непрерывный звук для настройки.
           """
           if not self.current_task:
               return
           
           mode = self.current_task["mode"]
           tonic_freq = self.current_task["tonic_freq"]
           interval_freq = self.current_task["interval_freq"]
           
           # Останавливаем текущий звук
           self.stop_sound()
           
           if mode == "harmonic":
               # В гармоническом режиме включаем тонику как фон
               self._start_harmonic_tonic()
               print("   🎵 ГАРМОНИЧЕСКИЙ РЕЖИМ: тоника звучит (фон)")
           else:
               # В мелодическом режиме включаем непрерывный звук для настройки
               self.sound_engine.set_frequency(interval_freq)
               self.sound_engine.set_volume(self.volume)
               self.sound_engine.timbre = Timbre(self.timbre_name)
               self.sound_engine.start()
               self.is_playing = True
               print(f"   🎵 МЕЛОДИЧЕСКИЙ РЕЖИМ: непрерывный звук {interval_freq:.1f} Гц")

   📊 Результаты:
      ✅ Обрабатывает гармонический режим
      ✅ Запускает звук для гармонического режима
      ✅ Упоминает тонику

   def _start_harmonic_tonic(self):
           """Запуск постоянного звучания тоники (гармонический режим)"""
           if not self.sound_engine or not self.current_task:
               return
           
           tonic_freq = self.current_task["tonic_freq"]
           
           self.sound_engine.stop()
           self.sound_engine.set_frequency(tonic_freq)
           self.sound_engine.set_volume(self.volume * 0.7)  # Немного тише для фона
           self.sound_engine.timbre = Timbre(self.timbre_name)
           self.sound_engine.start()
           self.is_harmonic_mode = True
           self.is_playing = True
           
           print(f"   🎵 ТОНИКА (фон): {tonic_freq:.1f} Гц")

   📊 Результаты:
      ✅ Запускает звук
      ✅ Устанавливает частоту
      ✅ Устанавливает флаг гармонического режима
